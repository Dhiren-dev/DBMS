EXPERIMENT 9: TRANSACTIONS AND CONCURRENCY CONTROL
Self-contained SQL practical • MySQL 8.0+ • Create database, tables and sample data included

-- EXPERIMENT 9: COMMIT, ROLLBACK and row locking
CREATE DATABASE IF NOT EXISTS college;
USE college;

CREATE TABLE department (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(40) NOT NULL
);

CREATE TABLE student (
  roll_no INT PRIMARY KEY,
  student_name VARCHAR(50) NOT NULL,
  age INT,
  dept_id INT,
  FOREIGN KEY (dept_id) REFERENCES department(dept_id)
) ENGINE = InnoDB;

INSERT INTO department VALUES (1, 'Computer'), (2, 'IT');
INSERT INTO student VALUES
(101, 'Amit', 20, 1), (102, 'Neha', 21, 2), (103, 'Rahul', 19, 1);

START TRANSACTION;
INSERT INTO student VALUES (104, 'Kiran', 20, 1);
UPDATE student SET age = 21 WHERE roll_no = 104;
SELECT * FROM student WHERE roll_no = 104;
COMMIT;
SELECT * FROM student WHERE roll_no = 104;

START TRANSACTION;
UPDATE student SET age = 25 WHERE roll_no = 104;
SELECT * FROM student WHERE roll_no = 104;
ROLLBACK;
SELECT * FROM student WHERE roll_no = 104;

-- Concurrency demo: run the following in Session 1
START TRANSACTION;
SELECT * FROM student WHERE roll_no = 104 FOR UPDATE;

-- Keep Session 1 transaction open.
-- In Session 2, run:
-- UPDATE student SET age = 22 WHERE roll_no = 104;
-- Session 2 waits because Session 1 has locked the row.
-- Back in Session 1, release the lock:
COMMIT;

-- Session 2 update can now finish. Check:
SELECT * FROM student WHERE roll_no = 104;
