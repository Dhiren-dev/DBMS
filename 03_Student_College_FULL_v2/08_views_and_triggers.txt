EXPERIMENT 8: VIEWS AND TRIGGERS
Self-contained SQL practical • MySQL 8.0+ • Create database, tables and sample data included

-- EXPERIMENT 8: Create a view and audit triggers
CREATE DATABASE IF NOT EXISTS college;
USE college;

CREATE TABLE department (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(40) NOT NULL
);

CREATE TABLE student (
  roll_no INT PRIMARY KEY,
  student_name VARCHAR(50) NOT NULL,
  age INT,
  dept_id INT,
  FOREIGN KEY (dept_id) REFERENCES department(dept_id)
);

INSERT INTO department VALUES (1, 'Computer'), (2, 'IT');

INSERT INTO student VALUES
(101, 'Amit', 20, 1), (102, 'Neha', 21, 2), (103, 'Rahul', 19, 1);

CREATE VIEW student_view AS
SELECT roll_no, student_name, dept_id FROM student;

SELECT * FROM student_view;

CREATE TABLE student_log (
  log_id INT AUTO_INCREMENT PRIMARY KEY,
  action_name VARCHAR(20) NOT NULL,
  roll_no INT,
  action_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

DELIMITER //

CREATE TRIGGER student_after_insert
AFTER INSERT ON student FOR EACH ROW
BEGIN
  INSERT INTO student_log (action_name, roll_no)
  VALUES ('INSERT', NEW.roll_no);
END//

CREATE TRIGGER student_after_update
AFTER UPDATE ON student FOR EACH ROW
BEGIN
  INSERT INTO student_log (action_name, roll_no)
  VALUES ('UPDATE', NEW.roll_no);
END//

CREATE TRIGGER student_after_delete
AFTER DELETE ON student FOR EACH ROW
BEGIN
  INSERT INTO student_log (action_name, roll_no)
  VALUES ('DELETE', OLD.roll_no);
END//

DELIMITER ;

INSERT INTO student VALUES (104, 'Sonal', 22, 1);
UPDATE student SET age = 23 WHERE roll_no = 104;
DELETE FROM student WHERE roll_no = 104;

SELECT * FROM student_log;
SHOW TRIGGERS;
