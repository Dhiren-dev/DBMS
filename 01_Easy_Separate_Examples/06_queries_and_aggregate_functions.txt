Complete independent practical • MySQL 8.0+ • Database, tables and sample records included

-- EXPERIMENT 6: SELECT and aggregate functions in an Office database
CREATE DATABASE IF NOT EXISTS office;
USE office;

CREATE TABLE employee (
  emp_id INT PRIMARY KEY,
  emp_name VARCHAR(40) NOT NULL,
  salary INT,
  department VARCHAR(30)
);

INSERT INTO employee VALUES
(1, 'Amit', 25000, 'Sales'),
(2, 'Neha', 35000, 'IT'),
(3, 'Ravi', 30000, 'Sales'),
(4, 'Kiran', 40000, 'HR'),
(5, 'Sonal', 45000, 'IT');

-- Basic SELECT
SELECT * FROM employee;
SELECT emp_name, salary FROM employee;

-- WHERE, AND, OR, BETWEEN, IN and LIKE
SELECT * FROM employee WHERE salary > 30000;
SELECT * FROM employee WHERE salary > 25000 AND department = 'IT';
SELECT * FROM employee WHERE department = 'IT' OR department = 'HR';
SELECT * FROM employee WHERE salary BETWEEN 25000 AND 40000;
SELECT * FROM employee WHERE department IN ('IT', 'Sales');
SELECT * FROM employee WHERE emp_name LIKE 'S%';

-- DISTINCT and ORDER BY
SELECT DISTINCT department FROM employee;
SELECT * FROM employee ORDER BY emp_name ASC;
SELECT * FROM employee ORDER BY salary DESC;

-- Aggregate functions
SELECT COUNT(*) AS total_employees FROM employee;
SELECT SUM(salary) AS total_salary FROM employee;
SELECT AVG(salary) AS average_salary FROM employee;
SELECT MIN(salary) AS lowest_salary FROM employee;
SELECT MAX(salary) AS highest_salary FROM employee;

-- GROUP BY and HAVING
SELECT department, COUNT(*) AS employee_count
FROM employee GROUP BY department;

SELECT department, AVG(salary) AS average_salary
FROM employee GROUP BY department
HAVING AVG(salary) > 30000;
