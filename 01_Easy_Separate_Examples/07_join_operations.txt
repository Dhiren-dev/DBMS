EXPERIMENT 7: JOIN OPERATIONS (CUSTOMER AND ORDERS)
Complete independent practical • MySQL 8.0+ • Database, tables and sample records included

-- EXPERIMENT 7: Join types using a simple Shop database
CREATE DATABASE IF NOT EXISTS shop;
USE shop;

CREATE TABLE customer (
  customer_id INT PRIMARY KEY,
  customer_name VARCHAR(40) NOT NULL,
  city VARCHAR(30)
);

CREATE TABLE orders (
  order_id INT PRIMARY KEY,
  customer_id INT,
  amount INT,
  FOREIGN KEY (customer_id) REFERENCES customer(customer_id)
);

INSERT INTO customer VALUES

-- Equijoin / INNER JOIN
SELECT customer.customer_name, orders.order_id, orders.amount
FROM customer INNER JOIN orders
ON customer.customer_id = orders.customer_id;

-- Non-equijoin (amount is compared with a range)
SELECT customer.customer_name, orders.amount
FROM customer JOIN orders
ON orders.amount > 400 AND orders.amount < 800;

-- Self-join: customers living in the same city
SELECT a.customer_name AS customer1, b.customer_name AS customer2,
  a.city
FROM customer a JOIN customer b
ON a.city = b.city AND a.customer_id < b.customer_id;

-- LEFT OUTER JOIN
SELECT customer.customer_name, orders.order_id
FROM customer LEFT JOIN orders
ON customer.customer_id = orders.customer_id;

-- RIGHT OUTER JOIN
SELECT customer.customer_name, orders.order_id
FROM customer RIGHT JOIN orders
ON customer.customer_id = orders.customer_id;

-- CROSS JOIN
SELECT customer.customer_name, orders.order_id
FROM customer CROSS JOIN orders;
