-- EXPERIMENT 7: Join operations
CREATE DATABASE IF NOT EXISTS college;
USE college;

CREATE TABLE department (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(40) NOT NULL
);

CREATE TABLE student (
  roll_no INT PRIMARY KEY,
  student_name VARCHAR(50) NOT NULL,
  age INT,
  dept_id INT,
  FOREIGN KEY (dept_id) REFERENCES department(dept_id)
);

-- Equijoin / INNER JOIN
SELECT student.student_name, department.dept_name
FROM student INNER JOIN department
ON student.dept_id = department.dept_id;

-- Non-equijoin
SELECT student.student_name, student.age, department.dept_name
FROM student JOIN department
ON student.age >= 20 AND department.dept_id <= 2;

-- Self-join: students in the same department
SELECT a.student_name AS student1, b.student_name AS student2, a.dept_id
FROM student a JOIN student b
ON a.dept_id = b.dept_id AND a.roll_no < b.roll_no;

-- LEFT OUTER JOIN
SELECT student.student_name, department.dept_name
FROM student LEFT JOIN department
ON student.dept_id = department.dept_id;

-- RIGHT OUTER JOIN
SELECT student.student_name, department.dept_name
FROM student RIGHT JOIN department
ON student.dept_id = department.dept_id;

-- CROSS JOIN
SELECT student.student_name, department.dept_name
FROM student CROSS JOIN department;
