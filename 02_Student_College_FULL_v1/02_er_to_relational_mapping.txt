EXPERIMENT 2: ER/EER TO RELATIONAL SCHEMA
Self-contained SQL practical • MySQL 8.0+ • Create database, tables and sample data included

-- EXPERIMENT 2: Map the ER/EER model to relations
CREATE DATABASE IF NOT EXISTS college;
USE college;

CREATE TABLE department (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(40) NOT NULL UNIQUE
);

CREATE TABLE student (
  roll_no INT PRIMARY KEY,
  student_name VARCHAR(50) NOT NULL,
  age INT,
  dept_id INT,
  FOREIGN KEY (dept_id) REFERENCES department(dept_id)
);

CREATE TABLE course (
  course_id INT PRIMARY KEY,
  course_name VARCHAR(50) NOT NULL,
  credits INT
);

-- M:N relationship between STUDENT and COURSE
CREATE TABLE enrollment (
  roll_no INT,
  course_id INT,
  marks INT,
  PRIMARY KEY (roll_no, course_id),
  FOREIGN KEY (roll_no) REFERENCES student(roll_no),
  FOREIGN KEY (course_id) REFERENCES course(course_id)
);
